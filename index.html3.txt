<script>
let selected = null;
let selectedPrice = 0;
let selectedPayment = null;

function selectCoin(el, amount, price) {
  document.querySelectorAll(".coin-card").forEach(c => c.classList.remove("active"));
  el.classList.add("active");
  selected = amount;
  selectedPrice = price;
  document.getElementById("total").innerText = amount;
}

function selectPayment(el, method) {
  document.querySelectorAll(".pay-option").forEach(c => c.classList.remove("active"));
  el.classList.add("active");
  selectedPayment = method;
}

function confirm() {
  const username = document.getElementById("username").value.trim();
  if (!username) {
    Swal.fire({ icon:'warning', title:'Warning', text:'Please enter your TikTok username!' });
    return;
  }
  if (!selected) {
    Swal.fire({ icon:'warning', title:'Warning', text:'Please select a coin package!' });
    return;
  }
  if (!selectedPayment) {
    Swal.fire({ icon:'warning', title:'Warning', text:'Please select a payment method!' });
    return;
  }

  Swal.fire({
    icon: 'success',
    title: '🎉 Purchase Successful!',
    html: `
      <p>You have successfully purchased <b>${selected}</b> coins for <b>@${username}</b>.</p>
      <p>💳 Payment Method: <b>${selectedPayment}</b></p>
      <p style="margin-top:8px; color:#666; font-size:14px;">
        ⏳ Please wait 5–15 minutes for the system to process your transaction.
      </p>
    `,
    confirmButtonText: 'OK',
    confirmButtonColor: '#16a34a'
  });
}

/* --- Loading + Tick xanh logic --- */
const usernameInput = document.getElementById("username");
const statusIcon = document.getElementById("statusIcon");
let typingTimer = null;

usernameInput.addEventListener("input", () => {
  clearTimeout(typingTimer);
  statusIcon.innerHTML = '<div class="spinner"></div>'; // Hiện loading
  typingTimer = setTimeout(() => {
    statusIcon.innerHTML = '<div class="tick">✔</div>'; // Sau 2s hiện tick xanh
  }, 2000);
});
